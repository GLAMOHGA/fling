local function isAlive(p)
    local c = p.Character
    if not c then return false end
    local h = c:FindFirstChildOfClass("Humanoid")
    return h and h.Health > 0
end

local function getSilentTarget()
    local center = Vector2.new(Cam.ViewportSize.X / 2, Cam.ViewportSize.Y / 2)
    local closest = getgenv().SilentAimFOV
    local target = nil
    for _, p in ipairs(Players:GetPlayers()) do
        if p == LP then continue end
        if not isAlive(p) then continue end
        local c = p.Character
        if not c then continue end
        local part = c:FindFirstChild(getgenv().SilentAimHitbox) or c:FindFirstChild("Head")
        if not part then continue end
        local pos, onScreen = Cam:WorldToViewportPoint(part.Position)
        if not onScreen or pos.Z <= 0 then continue end
        local dist = (Vector2.new(pos.X, pos.Y) - center).Magnitude
        if dist < closest then
            closest = dist
            target = part
        end
    end
    return target
end

local function setupSilentAim()
    local RS = game:GetService("ReplicatedStorage")
    local function findMod(n)
        for _, c in pairs(RS:GetDescendants()) do
            if c:IsA("ModuleScript") and c.Name == n then return c end
        end
        return nil
    end
    local MouseMod = findMod("Mouse") or findMod("Mouse_m")
    if not MouseMod then task.wait(1) setupSilentAim() return end
    local Mouse_m = require(MouseMod)
    local fakeMouse = setmetatable({}, {
        __index = function(t, k)
            if k == "WorldPosition" and getgenv().SilentAimEnabled then
                local tgt = getSilentTarget()
                if tgt then return tgt.Position end
            end
            return Mouse_m[k]
        end,
        __newindex = function(t, k, v) Mouse_m[k] = v end
    })
    local GunMod = findMod("Gun") or findMod("GunModule")
    if not GunMod then return end
    local gunTbl = require(GunMod)
    local origCreate = gunTbl.Create
    gunTbl.Create = function(self, arg)
        local res = origCreate(self, arg)
        local function patch(f, vis)
            if vis[f] then return end
            vis[f] = true
            local ups = getupvalues(f)
            for i, uv in pairs(ups) do
                if type(uv) == "table" and uv == Mouse_m then
                    setupvalue(f, i, fakeMouse)
                end
                if type(uv) == "function" then patch(uv, vis) end
            end
        end
        for k, v in pairs(res) do
            if type(v) == "function" then patch(v, {}) end
        end
        return res
    end
end

-- Wall Check Function
local function wallCheck(part)
    if not t1.vars.wallCheck then return true end
    if not LP.Character or not LP.Character:FindFirstChild("Head") then return true end
    local origin = Cam.CFrame.Position
    local dir = part.Position - origin
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = {LP.Character}
    local result = workspace:Raycast(origin, dir, params)
    if result then
        return result.Instance:IsDescendantOf(part.Parent)
    end
    return true
end

local function wallCheckPlayer(p)
    if not p.Character then return false end
    local head = p.Character:FindFirstChild("Head")
    if not head then return false end
    if not LP.Character or not LP.Character:FindFirstChild("Head") then return true end
    local origin = Cam.CFrame.Position
    local dir = head.Position - origin
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = {LP.Character}
    local result = workspace:Raycast(origin, dir, params)
    if result then
        return result.Instance:IsDescendantOf(p.Character)
    end
    return true
end

-- Snap Lines Update
local function updateSnapLines()
    if not t1.vars.snapLines or not t1.vars.targetHud or not t1.targetPart or not t1.targetPart.Parent then
        t1.snapLine.Visible = false
        return
    end
    local hum = t1.targetPart.Parent:FindFirstChild("Humanoid")
    if not hum or hum.Health <= 0 then
        t1.snapLine.Visible = false
        return
    end
    local pos, onScreen = Cam:WorldToViewportPoint(t1.targetPart.Position)
    if not onScreen then
        t1.snapLine.Visible = false
        return
    end
    local vp = Cam.ViewportSize
    t1.snapLine.From = Vector2.new(vp.X / 2, vp.Y)
    t1.snapLine.To = Vector2.new(pos.X, pos.Y)
    t1.snapLine.Color = t1.vars.snapColor
    t1.snapLine.Visible = true
end

-- Target HUD Update
local function updateTargetHUD()
    if not t1.vars.targetHud then
        t1.hudFrame.Visible = false
        return
    end
    if t1.targetPart and t1.targetPart.Parent then
        local char = t1.targetPart.Parent
        local p = Players:GetPlayerFromCharacter(char)
        if char and p then
            local hum = char:FindFirstChild("Humanoid")
            if hum and hum.Health > 0 then
                t1.hudFrame.Visible = true
                local visible = wallCheckPlayer(p)
                t1.hudName.TextColor3 = visible and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(255, 50, 50)
                t1.hudName.Text = p.Name
                if char ~= t1.lastTargetChar then
                    local ok, thumb = pcall(function()
                        return Players:GetUserThumbnailAsync(p.UserId, Enum.ThumbnailType.HeadShot, Enum.ThumbnailSize.Size100x100)
                    end)
                    if ok then t1.hudAvatar.Image = thumb end
                end
                local hp = hum.Health
                local ratio = math.clamp(hp / hum.MaxHealth, 0, 1)
                TS:Create(t1.hudHpFill, t1.tweenInfo, {Size = UDim2.new(ratio, 0, 1, 0)}):Play()
                t1.hudHpText.Text = string.format("%.0f", hp)
                if LP.Character and LP.Character:FindFirstChild("Head") and char:FindFirstChild("Head") then
                    t1.hudDist.Text = math.floor((LP.Character.Head.Position - char.Head.Position).Magnitude) .. "m"
                end
                t1.lastTargetChar = char
            else
                t1.hudFrame.Visible = false
            end
        else
            t1.hudFrame.Visible = false
        end
    else
        t1.hudFrame.Visible = false
    end
end

    if not t1.vars.infJump then return end
    if not LP.Character then return end

local function setupJumpBoost()
    if not t1.vars.jumpBoost then return end
    if not LP.Character then return end
    local hrp = LP.Character:FindFirstChild("HumanoidRootPart")
    local hum = LP.Character:FindFirstChild("Humanoid")
    if not hrp or not hum then return end
    if not LP.Character:GetAttribute("VOMA_JumpConn") then
        LP.Character:SetAttribute("VOMA_JumpConn", true)
        local conn = hum.StateChanged:Connect(function(_, new)
            if new == Enum.HumanoidStateType.Jumping and t1.vars.jumpBoost then
                task.defer(function()
                    if hrp and hrp.Parent then
                        local vel = hrp.Velocity
                        local extra = t1.vars.jumpPow - hum.JumpPower
                        if extra > 0 then
                            hrp.Velocity = Vector3.new(vel.X, vel.Y + extra * 0.5, vel.Z)
                        end
                    end
                end)
            end
        end)
        table.insert(t1.jumpConns, conn)
    end
end

LP.CharacterAdded:Connect(function(_)
    task.wait(0.5)
    if t1.vars.jumpBoost then setupJumpBoost() end
end)

-- Hitboxes (ОРИГИНАЛЬНАЯ ВЕРСИЯ БЕЗ БАГОВ)
local function applyHitboxes()
    if not t1.vars.hitboxes then return end
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= LP and p.Character then
            for part, enabled in pairs(t1.vars.hitboxParts) do
                if enabled then
                    local bp = p.Character:FindFirstChild(part)
                    if bp and bp:IsA("BasePart") then
                        local key = p.UserId .. "_" .. part
                        if not t1.origSizes[key] then
                            t1.origSizes[key] = bp.Size
                        end
                        bp.Size = Vector3.new(t1.vars.hitboxSize, t1.vars.hitboxSize, t1.vars.hitboxSize)
                        bp.Transparency = t1.vars.hitboxTrans
                        bp.Color = t1.vars.hitboxColor
                        bp.Material = Enum.Material.ForceField
                        bp.CanCollide = false
                        bp.Massless = true
                    end
                end
            end
        end
    end
end

local function removeHitboxes()
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= LP and p.Character then
            for _, part in pairs(t1.hitboxParts) do
                local bp = p.Character:FindFirstChild(part)
                if bp and bp:IsA("BasePart") then
                    local key = p.UserId .. "_" .. part
                    if t1.origSizes[key] then
                        bp.Size = t1.origSizes[key]
                        t1.origSizes[key] = nil
                    end
                    bp.Transparency = 0
                    bp.Material = Enum.Material.Plastic
                    bp.CanCollide = true
                    bp.Massless = false
                end
            end
        end
    end
end

    while true do
        if t1.vars.hitboxes then applyHitboxes() end
        task.wait(0.5)
    end
end)

local function spawnEffect(parent, color, typ)
    local att
    if typ == "Coom" then att = createFoam()
    elseif typ == "Slash" then att = createSlash()
    elseif typ == "Atomic Slash" then att = createAtomicSlash()
    elseif typ == "Thunder" then att = createThunder()
    elseif typ == "Electric" then att = createElectric()
    elseif typ == "Aura" then att = createAura()
    elseif typ == "Cosmic Explosion" then att = createCosmicExplosion()
    else return end
    att.Parent = parent or Cam
    for _, c in ipairs(att:GetChildren()) do
        if c:IsA("ParticleEmitter") then
            c.Color = ColorSequence.new(color or Color3.fromRGB(160, 96, 255))
            c.Enabled = true
        end
    end
    task.delay(2, function()
        for _, c in ipairs(att:GetChildren()) do
            if c:IsA("ParticleEmitter") then c.Enabled = false end
        end
        task.wait(1)
        if att and att.Parent then att:Destroy() end
    end)
end

-- Hit Chams
local function createHitChams(p)
    if not t1.vars.hitChams or not p or not p.Character or not p.Character:FindFirstChild("HumanoidRootPart") then return end
    
    local tempSizes = {}
    for _, part in pairs(p.Character:GetChildren()) do
        if part:IsA("BasePart") then
            tempSizes[part.Name] = part.Size
        end
    end
    
    if t1.vars.hitboxes then
        for _, part in pairs(p.Character:GetChildren()) do
            if part:IsA("BasePart") then
                local key = p.UserId .. "_" .. part.Name
                if t1.origSizes[key] then
                    part.Size = t1.origSizes[key]
                end
            end
        end
    end
    
    pcall(function() p.Character.Archivable = true end)
    local ok, clone = pcall(function() return p.Character:Clone() end)
    
    if t1.vars.hitboxes then
        for _, part in pairs(p.Character:GetChildren()) do
            if part:IsA("BasePart") then
                part.Size = tempSizes[part.Name] or part.Size
            end
        end
    end
    
    if not ok or not clone then return end
    
    clone.Name = "VOMA_HitChams"
    for _, d in pairs(clone:GetDescendants()) do
        if d:IsA("BillboardGui") or d:IsA("SurfaceGui") then d:Destroy() end
    end
    
    for _, c in ipairs(clone:GetChildren()) do
        if c:IsA("BasePart") then
            local keep = false
            for _, n in ipairs({"Head", "UpperTorso", "LowerTorso", "LeftUpperArm", "LeftLowerArm", "LeftHand", "RightUpperArm", "RightLowerArm", "RightHand", "LeftUpperLeg", "LeftLowerLeg", "LeftFoot", "RightUpperLeg", "RightLowerLeg", "RightFoot"}) do
                if n == c.Name then keep = true break end
            end
            if not keep then c:Destroy() end
        elseif not c:IsA("BasePart") then
            if c:IsA("Accessory") or c:IsA("Tool") or c:IsA("Shirt") or c:IsA("Pants") or c:IsA("Hat") or c:IsA("Script") or c:IsA("LocalScript") or c.Name == "face" then
                c:Destroy()
            end
        end
    end
    
    local hum = clone:FindFirstChildWhichIsA("Humanoid")
    if hum then hum:Destroy() end
    
    local mat = t1.vars.hitChamsMat == "ForceField" and Enum.Material.ForceField or Enum.Material.Neon
    for _, c in ipairs(clone:GetChildren()) do
        if c:IsA("BasePart") then
            c.CanCollide = false
            c.Anchored = true
            c.Transparency = t1.vars.hitChamsTrans
            c.Color = t1.vars.hitChamsColor
            c.Material = mat
            local face = c:FindFirstChild("face")
            if face then face:Destroy() end
        end
    end
    
    clone.Parent = workspace
    local ti = TweenInfo.new(t1.vars.hitChamsTime, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut, 0, false)
    for _, c in ipairs(clone:GetChildren()) do
        if c:IsA("BasePart") then
            TS:Create(c, ti, {Transparency = 1}):Play()
        end
    end
    
    task.delay(t1.vars.hitChamsTime + 0.5, function()
        if clone and clone.Parent then clone:Destroy() end
    end)
end

-- Hit Skeleton
local function createHitSkeleton(p)
    if not t1.vars.hitSkel or not p or not p.Character or not p.Character:FindFirstChild("HumanoidRootPart") then return end
    
    local char = p.Character
    local folder = Instance.new("Folder")
    folder.Name = "VOMA_Skeleton"
    folder.Parent = workspace
    
    for _, pair in ipairs({
        {"Head", "UpperTorso"}, {"UpperTorso", "LowerTorso"}, {"UpperTorso", "LeftUpperArm"}, {"LeftUpperArm", "LeftLowerArm"},
        {"LeftLowerArm", "LeftHand"}, {"UpperTorso", "RightUpperArm"}, {"RightUpperArm", "RightLowerArm"}, {"RightLowerArm", "RightHand"},
        {"LowerTorso", "LeftUpperLeg"}, {"LeftUpperLeg", "LeftLowerLeg"}, {"LeftLowerLeg", "LeftFoot"},
        {"LowerTorso", "RightUpperLeg"}, {"RightUpperLeg", "RightLowerLeg"}, {"RightLowerLeg", "RightFoot"}
    }) do
        local p1 = char:FindFirstChild(pair[1])
        local p2 = char:FindFirstChild(pair[2])
        if p1 and p2 then
            local mag = (p1.Position - p2.Position).Magnitude
            local mid = (p1.Position + p2.Position) / 2
            local part = Instance.new("Part")
            part.Anchored = true
            part.CanCollide = false
            part.Size = Vector3.new(0.2, 0.2, mag)
            part.CFrame = CFrame.lookAt(mid, p2.Position)
            part.Color = t1.vars.skelColor
            part.Material = Enum.Material.Neon
            part.Transparency = t1.vars.hitChamsTrans
            part.Parent = folder
        end
    end
    
    local ti = TweenInfo.new(t1.vars.hitChamsTime, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut)
    for _, c in ipairs(folder:GetChildren()) do
        TS:Create(c, ti, {Transparency = 1}):Play()
    end
    
    task.delay(t1.vars.hitChamsTime + 0.5, function()
        if folder and folder.Parent then folder:Destroy() end
    end)
end

-- Hook Gun Damage Event
local mt = getrawmetatable(game)
setreadonly(mt, false)
local oldNamecall = mt.__namecall

mt.__namecall = newcclosure(function(self, ...)
    local args = {...}
    local method = getnamecallmethod()
    
    if method == "FireServer" then
        local name = tostring(self)
        if name:find("Damage") or name:find("Hit") or name:find("Bullet") then
            local targetPlayer = args[1]
            
            local now = tick()
            if t1.lastHit[targetPlayer] and (now - t1.lastHit[targetPlayer]) < 0.15 then
                return oldNamecall(self, ...)
            end
            t1.lastHit[targetPlayer] = now
            
            task.spawn(function()
                if t1.vars.hitChams then createHitChams(targetPlayer) end
                if t1.vars.hitSkel then createHitSkeleton(targetPlayer) end
                if t1.vars.hitFX and targetPlayer and targetPlayer.Character then
                    local hrp = targetPlayer.Character:FindFirstChild("HumanoidRootPart")
                    if hrp then
                        spawnEffect(hrp, t1.vars.hitFXColor, t1.vars.hitFXType)
                    end
                end
            end)
        end
    end
    
    return oldNamecall(self, ...)
end)

setreadonly(mt, true)

-- Player Tracking
local function trackPlayer(p)
    if p == LP then return end
    local function onChar(c)
        task.wait(0.5)
        local hum = c:FindFirstChild("Humanoid")
        if not hum then return end
        t1.healthConns[p] = t1.healthConns[p] or {}
    end
    local conn = p.CharacterAdded:Connect(function(c) onChar(c) end)
    t1.healthConns[p] = t1.healthConns[p] or {}
    table.insert(t1.healthConns[p], conn)
    if p.Character then task.spawn(function() onChar(p.Character) end) end
end

for _, p in pairs(Players:GetPlayers()) do trackPlayer(p) end
Players.PlayerAdded:Connect(function(p)
    if t1.vars.esp then addESP(p) end
    if t1.vars.chams then addChams(p) end
    trackPlayer(p)
end)

Players.PlayerRemoving:Connect(function(p)
    removeESP(p)
    removeChams(p)
    t1.lastHit[p] = nil
    if t1.healthConns[p] then
        for _, c in ipairs(t1.healthConns[p]) do
            pcall(function() c:Disconnect() end)
        end
        t1.healthConns[p] = nil
    end
    for _, part in pairs(t1.hitboxParts) do
        t1.origSizes[p.UserId .. "_" .. part] = nil
    end
end)
