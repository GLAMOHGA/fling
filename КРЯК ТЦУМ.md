-- Функция для копирования в буфер
local function copyToClipboard(text)
    pcall(function()
        if syn and syn.setclipboard then
            syn.setclipboard(text)
        elseif setclipboard then
            setclipboard(text)
        elseif toclipboard then
            toclipboard(text)
        elseif SW and SW.SetClipboard then
            SW.SetClipboard(text)
        elseif Electron and Electron.setClipboard then
            Electron.setClipboard(text)
        end
    end)
end
