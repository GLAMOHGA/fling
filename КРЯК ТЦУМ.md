        function wrapped:Set(value)
            local payload = value
            if type(value) == "table" then
                if value.Options and control.Refresh then control:Refresh(value.Options) end
                payload = value.CurrentValue
                if payload == nil then payload = value.CurrentOption end
                if payload == nil then payload = value.Value end
                if payload == nil then payload = value.Color end
            end
            if payload ~= nil and control.Set then control:Set(payload) end
        end

        function wrapped:SetValue(value)
            if control.Set then control:Set(value) end
        end

        function wrapped:Select(value)
            if control.Set then control:Set(value) end
        end

        return wrapped
    end

    function Adapter:Notification(properties)
        WindUI:Notify({
            Title = properties.Title or "Flowering",
            Content = properties.Content or "",
            Duration = properties.Duration or 3,
            Icon = "bell",
        })
    end

    function Adapter:Destroy()
        if self.Window then pcall(function() self.Window:Destroy() end) end
        self.Window = nil
    end

    function Adapter:CreateWindow(properties)
        local rawWindow = WindUI:CreateWindow({
            Title = properties.Name or "Flowering",
            Author = properties.Subtitle or "t.me/floweringmint",
            Folder = "Flowering",
            Icon = "flower-2",

            local tab = rawWindow:Tab({
                Title = tabProperties.Name or "Tab",
                Icon = iconMap[tabProperties.Icon] or "circle",
            })

            function tab:CreateSection()
                return nil
            end

            function tab:CreateToggle(options)
                return wrapControl(tab:Toggle({
                    Title = options.Name,
                    Desc = options.Description,
                    Value = options.CurrentValue == true,
                    Callback = options.Callback,
                }), "Toggle")
            end

            function tab:CreateSlider(options)
                local range = options.Range or {0, 100}
                return wrapControl(tab:Slider({
                    Title = options.Name,
                    Desc = options.Description,
                    Step = options.Increment or 1,

                return wrapControl(tab:Input({
                    Title = options.Name,
                    Desc = options.Description,

local function decodeESPDatabase(raw)
    if type(raw) ~= "string" or raw == "" then return nil end
    local ok, database = pcall(HttpService.JSONDecode, HttpService, raw)
    if not ok or type(database) ~= "table" then return nil end
    if database.format ~= "flowering-esp-db-v1" or tonumber(database.version) ~= 1 then return nil end
    if type(database.items) ~= "table" or type(database.maps) ~= "table" then return nil end
    return database
end

local function loadESPDatabase()
    local remoteRaw = nil
    pcall(function()
        remoteRaw = game:HttpGet(ESP_DATABASE_URL, true)
    end)

    local remoteDatabase = decodeESPDatabase(remoteRaw)
    if remoteDatabase then
        if writefile then
            pcall(writefile, ESP_DATABASE_CACHE_FILE, remoteRaw)
        end
        return remoteDatabase
    end

    if readfile and (not isfile or isfile(ESP_DATABASE_CACHE_FILE)) then
        local ok, cachedRaw = pcall(readfile, ESP_DATABASE_CACHE_FILE)
        if ok then
            local cachedDatabase = decodeESPDatabase(cachedRaw)
            if cachedDatabase then return cachedDatabase end
        end
    end

    warn("[Flowering] ESP database unavailable; using game configs only")
    return {items = {}, maps = {}, fallbackAliases = {}, fallbackItems = {}, clothingOverrides = {}}
end

local function decodeClothingTemplateDatabase(raw)
    if type(raw) ~= "string" or raw == "" then return nil end
    local ok, database = pcall(HttpService.JSONDecode, HttpService, raw)
    if not ok or type(database) ~= "table" then return nil end
    if database.format ~= "flowering-clothing-templates-v1" or tonumber(database.version) ~= 1 then return nil end
    if type(database.shirts) ~= "table" or type(database.pants) ~= "table" then return nil end
    return database
end

local function readClothingTemplateFile(path)
    if type(readfile) ~= "function" then return nil end
    if type(isfile) == "function" and not isfile(path) then return nil end
    local ok, raw = pcall(readfile, path)
    if not ok then return nil end
    return decodeClothingTemplateDatabase(raw), raw
end

local function loadClothingTemplateDatabase()
    local localDatabase = readClothingTemplateFile(CLOTHING_TEMPLATE_DATABASE_FILE)
    if localDatabase then return localDatabase end

    local remoteRaw = nil
    pcall(function()
        remoteRaw = game:HttpGet(CLOTHING_TEMPLATE_DATABASE_URL, true)
    end)
    local remoteDatabase = decodeClothingTemplateDatabase(remoteRaw)
    if remoteDatabase then
        if type(writefile) == "function" then
            pcall(writefile, CLOTHING_TEMPLATE_DATABASE_CACHE_FILE, remoteRaw)
        end
        return remoteDatabase
    end

    local cachedDatabase = readClothingTemplateFile(CLOTHING_TEMPLATE_DATABASE_CACHE_FILE)
    return cachedDatabase or {shirts = {}, pants = {}}
end

local function getNameTag()
    local bb = nil
    local pg = player:FindFirstChild("PlayerGui")
    if pg and pg:FindFirstChild("LocalResellerNameTag") then
        bb = pg:FindFirstChild("LocalResellerNameTag")
    elseif player.Character and player.Character:FindFirstChild("Head") then

local function applyNickname()
    local bb, nameLbl = getNameTag()
    if not bb or not nameLbl then return end
    if originalNameText == nil then originalNameText = nameLbl.Text end
    if originalNamePos == nil then originalNamePos = nameLbl.Position end

    local displayName = (customNickname ~= "" and NICK_ENABLED) and customNickname or originalNameText
    nameLbl.Text = displayName
    nameLbl.TextColor3 = Color3.fromRGB(255, 255, 255)

    local container = nameLbl.Parent

    local hasAnyPin = false
    for _, enabled in pairs(SELECTED_PINS) do
        if enabled then hasAnyPin = true break end
    end

    if hasAnyPin then
        if not pinIconsContainer or not pinIconsContainer.Parent then
            local iconFrame = Instance.new("Frame")
            iconFrame.Name = "TsumPinsContainer"
            iconFrame.BackgroundTransparency = 1
            iconFrame.ZIndex = (nameLbl.ZIndex or 1) + 1
            iconFrame.Parent = container
            pinIconsContainer = iconFrame

            local layout = Instance.new("UIListLayout")
            layout.FillDirection = Enum.FillDirection.Horizontal
            layout.HorizontalAlignment = Enum.HorizontalAlignment.Center
            layout.VerticalAlignment = Enum.VerticalAlignment.Center
            layout.Padding = UDim.new(0, 3)
            layout.Parent = iconFrame
        end

        for _, child in ipairs(pinIconsContainer:GetChildren()) do
            if child:IsA("ImageLabel") then child:Destroy() end
        end

        local selectedCount = 0
        for _, enabled in pairs(SELECTED_PINS) do if enabled then selectedCount = selectedCount + 1 end end

        local iconOrder = {"Developer","YouTube","TikTok","Moderator","Verify","Star","VIP","Admin","Owner"}
        for _, pinName in ipairs(iconOrder) do
            if SELECTED_PINS[pinName] then
                local img = Instance.new("ImageLabel")
                img.Name = "Pin_" .. pinName
                img.BackgroundTransparency = 1
                img.ScaleType = Enum.ScaleType.Fit
                img.Image = PIN_TYPES[pinName]
                img.Size = UDim2.new(0, 18, 0, 18)
                img.ZIndex = (nameLbl.ZIndex or 1) + 1
                img.Parent = pinIconsContainer
            end
        end

        local frameWidth = (selectedCount * 18) + ((selectedCount - 1) * 3)
        pinIconsContainer.Size = UDim2.fromOffset(frameWidth, 18)
        pinIconsContainer.Position = UDim2.new(originalNamePos.X.Scale, originalNamePos.X.Offset, originalNamePos.Y.Scale, originalNamePos.Y.Offset - 22)
        pinIconsContainer.Visible = true

        nameLbl.Position = originalNamePos

        nameLbl.Position = originalNamePos
    end

local function getContainerTime()
    local success, result = pcall(function()
        for _, desc in ipairs(player.PlayerGui:GetDescendants()) do
            if desc:IsA("TextLabel") and desc.Visible then
                local text = desc.Text:lower()
                if string.find(text, "контейнер") or string.find(text, "container") or 
                   string.find(text, "кейс") or string.find(text, "ящик") or 
                   string.find(text, "airdrop") or string.find(text, "дроп") then
                    
                    local m, s = string.match(text, "(%d+):(%d+)")
                    if m and s then
                        return tonumber(m) * 60 + tonumber(s)
                    end
                    local mins = string.match(text, "(%d+)%s*мин") or string.match(text, "(%d+)%s*min") or string.match(text, "(%d+)%s*m")
                    if mins then
                        return tonumber(mins) * 60
                    end
                    
                    local parent = desc.Parent
                    if parent then
                        for _, sibling in ipairs(parent:GetChildren()) do
                            if sibling:IsA("TextLabel") and sibling.Visible and sibling ~= desc then
                                local sibText = sibling.Text:lower()
                                local sm, ss = string.match(sibText, "(%d+):(%d+)")
                                if sm and ss then
                                    return tonumber(sm) * 60 + tonumber(ss)
                                end
                                local smins = string.match(sibText, "^%s*(%d+)%s*мин") or string.match(sibText, "^%s*(%d+)%s*min") or string.match(sibText, "^%s*(%d+)%s*m%s*$")
                                if smins then
                                    return tonumber(smins) * 60
                                end
                            end
                        end
                    end
                end
            end
        end
    end)
    if success and result then return result end
    return nil
end

                game:HttpGet(string.format("https://games.roblox.com/v1/games/%d/servers/Public?sortOrder=Asc&limit=100", placeId))
            )
        end)
        if success and result and result.data then
            for _, server in ipairs(result.data) do
                if server.playing < server.maxPlayers and server.id ~= game.JobId then
                    table.insert(servers, server.id)
                end
            end
            if #servers > 0 then
                local randomServer = servers[math.random(1, #servers)]
                notify("Телепорт", "Телепортация на случайный сервер...", 2)
                TeleportService:TeleportToPlaceInstance(placeId, randomServer, player)
            else
                notify("Ошибка", "Не найдено доступных серверов!", 4)
                isServerHopping = false
            end
        else
            notify("Ошибка", "Не удалось получить список серверов!", 4)
            isServerHopping = false
        end
    end)
end

local function normalizeNPCName(value)
    return tostring(value or ""):lower():gsub("[%s%p_]+", "")
end

local function getNPCContainer(instance)
    if not instance then return nil end
    if instance:IsA("Model") then return instance end
    if instance:IsA("BasePart") then
        return instance:FindFirstAncestorOfClass("Model") or instance
    end
    if instance:IsA("Folder") then
        return instance:FindFirstChildWhichIsA("Model", true)
            or instance:FindFirstChildWhichIsA("BasePart", true)
    end
    return instance:FindFirstAncestorOfClass("Model")
end

local function findSelectedNPC()
    local definition = NPC_TARGETS[SELECTED_NPC_TARGET]
    if not definition then return nil, nil end

    for _, name in ipairs(definition.names) do
        local exact = workspace:FindFirstChild(name, true)
        local container = getNPCContainer(exact)
        if container then return container, definition end
    end

    local wantedNames = {}
    for _, name in ipairs(definition.names) do
        wantedNames[normalizeNPCName(name)] = true
    end
    wantedNames[normalizeNPCName(definition.displayName)] = true

    for _, instance in ipairs(workspace:GetDescendants()) do
        if instance:IsA("Model") or instance:IsA("BasePart") then
            local normalized = normalizeNPCName(instance.Name)
            if wantedNames[normalized] then
                return getNPCContainer(instance), definition
            end
        elseif instance:IsA("ProximityPrompt") then
            local promptText = normalizeNPCName(tostring(instance.ObjectText) .. " " .. tostring(instance.ActionText))
            for wanted in pairs(wantedNames) do
                if wanted ~= "" and string.find(promptText, wanted, 1, true) then
                    local container = getNPCContainer(instance.Parent)
                    if container then return container, definition end
                end
            end
        end
    end

    return nil, definition
end

local function getNPCTargetPart(npc)
    if not npc then return nil end
    if npc:IsA("BasePart") then return npc end
    return npc:FindFirstChild("HumanoidRootPart", true)
        or npc.PrimaryPart
        or npc:FindFirstChild("Head", true)
        or npc:FindFirstChildWhichIsA("BasePart", true)
end

local function teleportToSelectedNPC()
    task.spawn(function()
        local npc, definition = findSelectedNPC()
        local displayName = definition and definition.displayName or tostring(SELECTED_NPC_TARGET)
        local fixedDestination = definition and definition.safeCFrame
        if not npc and not fixedDestination then
            notify("NPC Teleport", "Не найден NPC: " .. displayName, 4)
            return
        end

        local character = player.Character or player.CharacterAdded:Wait()
        local hrp = character:FindFirstChild("HumanoidRootPart") or character:WaitForChild("HumanoidRootPart", 5)
        local humanoid = character:FindFirstChildOfClass("Humanoid")
        local targetPart = getNPCTargetPart(npc)
        if not hrp or not humanoid or (not targetPart and not fixedDestination) then
            notify("NPC Teleport", "Не найдена точка телепорта к " .. displayName, 4)
            return
        end

        local originalAutoRotate = humanoid.AutoRotate
        local flyWasEnabled = FLY_ENABLED
        local teleportCollisionDefaults = {}
        FLY_ENABLED = false

        local ok, err = xpcall(function()
            RunService.Heartbeat:Wait()
            humanoid.AutoRotate = false

            for _, instance in ipairs(character:GetDescendants()) do
                if instance:IsA("BasePart") then
                    teleportCollisionDefaults[instance] = instance.CanCollide
                    instance.CanCollide = false
                end
            end

            local rayParams = RaycastParams.new()
            rayParams.FilterType = Enum.RaycastFilterType.Exclude
            rayParams.FilterDescendantsInstances = npc and {character, npc} or {character}

            local function groundPosition(horizontal)
                -- Start just above the NPC's own level so an upper floor or roof
                -- cannot be mistaken for the floor of an underground room.
                local rayOrigin = Vector3.new(horizontal.X, targetPart.Position.Y + 4, horizontal.Z)
                local ground = workspace:Raycast(rayOrigin, Vector3.new(0, -60, 0), rayParams)
                if ground then
                    local standingHeight = humanoid.HipHeight + (hrp.Size.Y * 0.5) + 0.35
                    return Vector3.new(horizontal.X, ground.Position.Y + standingHeight, horizontal.Z)
                end
                return nil
            end

            local candidates = {}
            if fixedDestination then
                table.insert(candidates, fixedDestination.Position)
            else
                local look = Vector3.new(targetPart.CFrame.LookVector.X, 0, targetPart.CFrame.LookVector.Z)
                if look.Magnitude < 0.1 then look = Vector3.new(0, 0, -1) else look = look.Unit end
                local right = Vector3.new(-look.Z, 0, look.X)
                local center = targetPart.Position
                local diagonalA = (look + right).Unit
                local diagonalB = (look - right).Unit
                local directions = {
                    look, -look, right, -right,
                    diagonalA, -diagonalA, diagonalB, -diagonalB,
                }
                for _, radius in ipairs({4.5, 7.5}) do
                    for _, direction in ipairs(directions) do
                        local candidate = groundPosition(center + direction * radius)
                        if candidate then table.insert(candidates, candidate) end
                    end
                end
            end

            local destination = nil
            local shortestDistance = math.huge
            for _, candidate in ipairs(candidates) do
                local distance = (candidate - hrp.Position).Magnitude
                if distance < shortestDistance then
                    shortestDistance = distance
                    destination = candidate
                end
            end

            if not destination then error("ground point not found near " .. tostring(npc:GetFullName())) end

            while not SCRIPT_UNLOADED and character.Parent and hrp.Parent and humanoid.Health > 0 do
                local offset = destination - hrp.Position
                if offset.Magnitude <= 0.15 then break end
                local dt = RunService.Heartbeat:Wait()
                local stepLength = math.min(offset.Magnitude, 90 * math.clamp(dt, 1 / 240, 1 / 20))
                local nextPosition = hrp.Position + offset.Unit * stepLength
                character:PivotTo(CFrame.lookAt(nextPosition, nextPosition + offset.Unit))
                hrp.AssemblyLinearVelocity = Vector3.zero
                hrp.AssemblyAngularVelocity = Vector3.zero
            end

            if fixedDestination then
                hrp.CFrame = fixedDestination
            else
                local currentPosition = hrp.Position
                hrp.CFrame = CFrame.lookAt(currentPosition, Vector3.new(targetPart.Position.X, currentPosition.Y, targetPart.Position.Z))
            end
            hrp.AssemblyLinearVelocity = Vector3.zero
            hrp.AssemblyAngularVelocity = Vector3.zero
        end, debug.traceback)

        for part, originalCanCollide in pairs(teleportCollisionDefaults) do
            pcall(function()
                if part.Parent then part.CanCollide = originalCanCollide end
            end)
        end

        pcall(function()
            humanoid.AutoRotate = originalAutoRotate
            if humanoid.Health > 0 then
                humanoid.Sit = false
                humanoid.PlatformStand = false
                humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
                task.wait(0.1)
                humanoid:ChangeState(Enum.HumanoidStateType.Running)
            end
        end)
        FLY_ENABLED = not SCRIPT_UNLOADED and flyWasEnabled or false

        if ok then
            notify("NPC Teleport", "Телепорт к " .. displayName, 3)
        else
            warn("[Flowering] NPC teleport error: " .. tostring(err))
            notify("NPC Teleport", "Ошибка телепорта к " .. displayName, 4)
        end
    end)
end

local function fmtPrice(price, spawnChance, economyProfile)
    local parts = {}
    if price ~= nil then
        local priceStr = tostring(math.floor(price)):reverse():gsub("(%d%d%d)", "%1 "):reverse():gsub("^ ", "")
        table.insert(parts, '<font size="13"><b>$' .. priceStr .. '</b></font>')
    end
    if spawnChance ~= nil then
        table.insert(parts, '<font size="10" color="#FFD700">Шанс: ' .. tostring(spawnChance) .. '%</font>')
    end
    if economyProfile then
        local typeStr
        if economyProfile == "jackpot" then
            typeStr = '<font size="11" color="#FFD700">[JACKPOT]</font>'
        else
            local color = getTypeColor(economyProfile)
            typeStr = '<font size="11" color="' .. color .. '">[' .. string.upper(economyProfile) .. ']</font>'
        end
        table.insert(parts, typeStr)
    end
    return table.concat(parts, "\n")
end

local NoclipState = {
    connection = nil,
    autoDepth = 0,
    defaults = setmetatable({}, {__mode = "k"}),
}

local function restoreNoclipParts()
    for part, originalCanCollide in pairs(NoclipState.defaults) do
        pcall(function()
            if part and part.Parent then part.CanCollide = originalCanCollide end
        end)
    end
    table.clear(NoclipState.defaults)
end


local function ensureNoclipConnection()
    if NoclipState.connection then return end
    NoclipState.connection = RunService.Stepped:Connect(function()
        if SCRIPT_UNLOADED then return end
        local enabled = NOCLIP_ENABLED or NoclipState.autoDepth > 0
        local character = player.Character
        if not enabled or not character then return end
        for _, part in ipairs(character:GetDescendants()) do
            if part:IsA("BasePart") then
                if NoclipState.defaults[part] == nil then
                    NoclipState.defaults[part] = part.CanCollide
                end
                part.CanCollide = false
            end
        end
    end)
end

local function startNoclip()
    NoclipState.autoDepth = NoclipState.autoDepth + 1
    ensureNoclipConnection()
end

local function stopNoclip(force)
    if force then
        NoclipState.autoDepth = 0
        NOCLIP_ENABLED = false
    else
        NoclipState.autoDepth = math.max(0, NoclipState.autoDepth - 1)
    end
    if NOCLIP_ENABLED or NoclipState.autoDepth > 0 then return end
    if NoclipState.connection then
        NoclipState.connection:Disconnect()
        NoclipState.connection = nil
    end
    restoreNoclipParts()
end

local function setNoclipEnabled(enabled)
    NOCLIP_ENABLED = enabled == true
    if NOCLIP_ENABLED then
        ensureNoclipConnection()
    elseif NoclipState.autoDepth <= 0 then
        stopNoclip(false)
    end
end

local function checkLineOfSight(startPos, endPos)
    local direction = endPos - startPos
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    local filter = {player.Character}
    local droppedFolder = workspace:FindFirstChild("DroppedItems")
    if droppedFolder then table.insert(filter, droppedFolder) end
    params.FilterDescendantsInstances = filter
    
    local result = workspace:Raycast(startPos, direction, params)
    if result then
        if result.Instance.CanCollide and result.Instance.Transparency < 0.9 then
            local isObstacle = true
            local p = result.Instance
            while p and p ~= workspace do
                if p:IsA("Accessory") or p:IsA("Tool") or Players:GetPlayerFromCharacter(p) then
                    isObstacle = false
                    break
                end
                p = p.Parent
            end
            if isObstacle then
                return false
            end
        end
    end
    return true
end

local function simplifyPath(waypoints)
    if #waypoints <= 2 then return waypoints end
    local simplified = {waypoints[1]}
    local currentIdx = 1
    
    while not SCRIPT_UNLOADED and currentIdx < #waypoints do
        local nextIdx = currentIdx + 1
        for i = #waypoints, currentIdx + 2, -1 do
            local sPos = waypoints[currentIdx].Position + Vector3.new(0, 2, 0)
            local ePos = waypoints[i].Position + Vector3.new(0, 2, 0)
            if checkLineOfSight(sPos, ePos) then
                nextIdx = i
                break
            end
        end
        table.insert(simplified, waypoints[nextIdx])
        currentIdx = nextIdx
    end
    return simplified
end

local function tweenTo(hrp, targetPos, speed)
    local startPos = hrp.Position
    local diff = targetPos - startPos
    local distance = diff.Magnitude
    if distance < 0.05 then 
        hrp.CFrame = CFrame.new(targetPos)
        return true 
    end
    local duration = distance / speed
    local startTime = tick()
    
    local success = true
    while not SCRIPT_UNLOADED and tick() - startTime < duration do
        if not player.Character or not hrp.Parent then
            success = false
            break
        end
        local alpha = (tick() - startTime) / duration
        local lerpedPos = startPos:Lerp(targetPos, alpha)
        local targetLook = Vector3.new(targetPos.X, lerpedPos.Y, targetPos.Z)
        if (targetLook - lerpedPos).Magnitude > 0.1 then
            hrp.CFrame = CFrame.lookAt(lerpedPos, targetLook)
        else
            hrp.CFrame = CFrame.new(lerpedPos)
        end
        RunService.Heartbeat:Wait()
    end
    
    if success and player.Character and hrp.Parent then
        hrp.CFrame = CFrame.new(targetPos)
    end
    return success
end

local function tweenFly(targetPos, speed)
    local char = player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return false end
    
    speed = speed or 30
    hrp.Anchored = true
    local success = tweenTo(hrp, targetPos, speed)
    hrp.Anchored = false
    return success
end

local function flySafe(targetPos, speed)
    local char = player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return false end
    
    speed = speed or 30
    hrp.Anchored = true
    
    local startPos = hrp.Position
    local horizontalDist = Vector3.new(targetPos.X - startPos.X, 0, targetPos.Z - startPos.Z).Magnitude
    if horizontalDist < 6 then
        local ok = tweenTo(hrp, targetPos, speed)
        hrp.Anchored = false
        return ok
    end
    
    local safeY = 26
    if startPos.Y > safeY then safeY = startPos.Y end
    if targetPos.Y > safeY then safeY = targetPos.Y end
    
    local upPos = Vector3.new(startPos.X, safeY, startPos.Z)
    local ok1 = tweenTo(hrp, upPos, speed)
    if not ok1 then 
        hrp.Anchored = false 
        return false 
    end
    
    local midPos = Vector3.new(targetPos.X, safeY, targetPos.Z)
    local ok2 = tweenTo(hrp, midPos, speed)
    if not ok2 then 
        hrp.Anchored = false 
        return false 
    end
    
    local ok3 = tweenTo(hrp, targetPos, speed)
    hrp.Anchored = false
    return ok3
end

local function flyWithPath(targetPos, speed, stopDistance)
    local char = player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return false end
    
    stopDistance = stopDistance or 0
    speed = speed or 30
    local startPos = hrp.Position
    
    if (targetPos - startPos).Magnitude < 8 then
        local dir = (targetPos - startPos).Unit
        local stopPos = targetPos - dir * stopDistance
        if (stopPos - startPos).Magnitude < 0.1 then return true end
        return tweenFly(stopPos, speed)
    end
    
    local path = PathfindingService:CreatePath({
        AgentRadius = 2.2,
        AgentHeight = 5.0,
        AgentCanJump = true,
        WaypointSpacing = 5,
    })
    
    local computeSuccess = false
    pcall(function()
        path:ComputeAsync(startPos, targetPos)
        computeSuccess = (path.Status == Enum.PathStatus.Success)
    end)
    
    if not computeSuccess then
        local dir = (startPos - targetPos).Unit
        local offsetPos = targetPos + dir * 5.5
        pcall(function()
            path:ComputeAsync(startPos, offsetPos)
            computeSuccess = (path.Status == Enum.PathStatus.Success)
        end)
    end
    
    if computeSuccess then
        local waypoints = path:GetWaypoints()
        
        hrp.Anchored = true
        local success = true
        for i, waypoint in ipairs(waypoints) do
            if isAutobuying and not AUTO_BUY_ENABLED then 
                success = false
                break 
            end
            if not player.Character or not hrp.Parent then 
                success = false
                break 
            end
            
            local wPos = waypoint.Position + Vector3.new(0, 2, 0)
            
            if i == #waypoints and stopDistance > 0 then
                local dir = (targetPos - hrp.Position).Unit
                wPos = targetPos - dir * stopDistance
            end
            
            local ok = tweenTo(hrp, wPos, speed)
            if not ok then 
                success = false
                break 
            end
        end
        
        if success then
            local dir = (targetPos - hrp.Position).Unit
            local finalPos = targetPos - dir * stopDistance
            tweenTo(hrp, finalPos, speed)
        end
        hrp.Anchored = false
        return success
    end

    local dir = (targetPos - startPos).Unit
    local finalPos = targetPos - dir * stopDistance
    return tweenFly(finalPos, speed)
end

local function tryWalkUnstick(hum, hrp, targetPos)
    if not hum or not hrp or not hrp.Parent then return false end

    local toTarget = targetPos - hrp.Position
    if toTarget.Magnitude < 0.1 then return false end

    local forward = Vector3.new(toTarget.X, 0, toTarget.Z)
    if forward.Magnitude < 0.1 then
        forward = Vector3.new(hrp.CFrame.LookVector.X, 0, hrp.CFrame.LookVector.Z)
    end
    if forward.Magnitude < 0.1 then return false end
    forward = forward.Unit

    local right = Vector3.new(-forward.Z, 0, forward.X)
    local candidates = {
        hrp.Position + right * 7 + forward * 4,
        hrp.Position - right * 7 + forward * 4,
        hrp.Position + right * 10,
        hrp.Position - right * 10,
        hrp.Position - forward * 4 + right * 6,
        hrp.Position - forward * 4 - right * 6,
    }

    local bestPos = nil
    local bestDist = math.huge
    for _, pos in ipairs(candidates) do
        local clear = checkLineOfSight(hrp.Position + Vector3.new(0, 2, 0), pos + Vector3.new(0, 2, 0))
        local distToTarget = (pos - targetPos).Magnitude
        if clear and distToTarget < bestDist then
            bestDist = distToTarget
            bestPos = pos
        end
    end

    if not bestPos then return false end

    hum.Jump = true
    hum:MoveTo(bestPos)
    local startTime = tick()
    while tick() - startTime < 0.9 do
        if not hum.Parent or not hrp.Parent then return false end
        if (hrp.Position - bestPos).Magnitude < 3 then
            return true
        end
        task.wait(0.04)
    end
    return true
end

local function walkTo(targetPos, speed, stopDistance)
    local char = player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hrp or not hum then return false end
    
    stopDistance = stopDistance or 4
    if (hrp.Position - targetPos).Magnitude <= stopDistance then
        return true
    end
    
    speed = speed or 30
    local originalSpeed = hum.WalkSpeed
    hum.WalkSpeed = speed
    
    local startPos = hrp.Position
    local pathTarget = targetPos
    local targetOffset = targetPos - startPos
    if stopDistance > 0 and targetOffset.Magnitude > stopDistance then
        pathTarget = targetPos - targetOffset.Unit * stopDistance
    end
    local path = PathfindingService:CreatePath({
        AgentRadius = 2.0,
        AgentHeight = 5.0,
        AgentCanJump = true,
        WaypointSpacing = 4,
    })
    
    local computeSuccess = false
    pcall(function()
        path:ComputeAsync(startPos, pathTarget)
        computeSuccess = (path.Status == Enum.PathStatus.Success)
    end)
    
    if computeSuccess then
        local waypoints = path:GetWaypoints()
        for i, waypoint in ipairs(waypoints) do
            if isAutobuying and not AUTO_BUY_ENABLED then 
                hum.WalkSpeed = originalSpeed
                return false 
            end
            if not player.Character or not hrp.Parent or not hum.Parent then 
                hum.WalkSpeed = originalSpeed
                return false 
            end
            
            if waypoint.Action == Enum.PathWaypointAction.Jump then
                hum.Jump = true
            end
            
            hum:MoveTo(waypoint.Position)
            
            local reached = false
            local connection
            connection = hum.MoveToFinished:Connect(function()
                reached = true
                if connection then connection:Disconnect(); connection = nil end
            end)
            
            local lastPos = hrp.Position
            local lastPosCheck = tick()
            local startTime = tick()
            while not SCRIPT_UNLOADED and not reached do
                if not player.Character or not hrp.Parent or not hum.Parent then
                    if connection then connection:Disconnect() end
                    hum.WalkSpeed = originalSpeed
                    return false
                end
                
                if (hrp.Position - targetPos).Magnitude <= stopDistance then
                    if connection then connection:Disconnect() end
                    hum:MoveTo(hrp.Position)
                    hum.WalkSpeed = originalSpeed
                    return true
                end
                
                local now = tick()
                if now - startTime > 1.5 then
                    pcall(function()
                        tryWalkUnstick(hum, hrp, targetPos)
                    end)
                    break
                end
                
                if now - lastPosCheck > 0.3 then
                    local distMoved = (hrp.Position - lastPos).Magnitude
                    if distMoved < 1.0 then
                        hum.Jump = true
                    end
                    lastPos = hrp.Position
                    lastPosCheck = now
                end
                
                task.wait(0.02)
            end
            if connection then
                connection:Disconnect()
            end
        end
        hum.WalkSpeed = originalSpeed
        return true
    else
        hum:MoveTo(pathTarget)
        
        local reached = false
        local connection
        connection = hum.MoveToFinished:Connect(function()
            reached = true
            if connection then connection:Disconnect(); connection = nil end
        end)
        
        local lastPos = hrp.Position
        local lastPosCheck = tick()
        local startTime = tick()
        while not SCRIPT_UNLOADED and not reached do
            if not player.Character or not hrp.Parent or not hum.Parent then
                if connection then connection:Disconnect() end
                hum.WalkSpeed = originalSpeed
                return false
            end
            
            if (hrp.Position - targetPos).Magnitude <= stopDistance then
                if connection then connection:Disconnect() end
                hum:MoveTo(hrp.Position)
                hum.WalkSpeed = originalSpeed
                return true
            end
            local now = tick()
            if now - startTime > 5.0 then
                pcall(function()
                    tryWalkUnstick(hum, hrp, targetPos)
                end)
                break
            end
            if now - lastPosCheck > 0.4 then
                local distMoved = (hrp.Position - lastPos).Magnitude
                if distMoved < 1.0 then
                    hum.Jump = true
                end
                lastPos = hrp.Position
                lastPosCheck = now
            end
            task.wait(0.05)
        end
        if connection then connection:Disconnect() end
        hum.WalkSpeed = originalSpeed
        return true
    end
end

local function travelTo(targetPos, speed, stopDistance)
    if AUTO_BUY_METHOD == "Walk" then
        return walkTo(targetPos, speed, stopDistance)
    else
        startNoclip()
        RunService.Stepped:Wait()
        local ok, success = pcall(flyWithPath, targetPos, speed, stopDistance)
        stopNoclip()
        if not ok then
            warn("[Flowering Auto Buy] Ошибка полёта: " .. tostring(success))
            return false
        end
        return success == true
    end
end

local function extractItems(tbl, outList)
    if type(tbl) == "table" then
        if tbl.name and (tbl.id or tbl.assetId or tbl.customModelName or tbl.meshId) then
            table.insert(outList, tbl)
        else
            for k, v in pairs(tbl) do
                if type(v) == "table" and k ~= "Categories" and k ~= "RARITIES" then
                    extractItems(v, outList)
                end
            end
        end
    end
end

local function normalizeContentId(value)
    local normalized = tostring(value or ""):lower():gsub("\\", "")
    local numericId = normalized:match("[?&]id=(%d+)")
        or normalized:match("rbxassetid://(%d+)")
        or normalized:match("^%s*(%d+)%s*$")
    return numericId or normalized
end

local function getVisualKey(inst)
    if not inst then return nil end
    local meshId = nil
    local textureId = nil
    if inst:IsA("MeshPart") then
        meshId = inst.MeshId
        textureId = inst.TextureID
        if not textureId or textureId == "" then
            local surface = inst:FindFirstChildOfClass("SurfaceAppearance")
            if surface then textureId = surface.ColorMap end
        end
    elseif inst:IsA("SpecialMesh") then
        meshId = inst.MeshId
        textureId = inst.TextureId
    end
    meshId = normalizeContentId(meshId)
    textureId = normalizeContentId(textureId)
    if meshId == "" or textureId == "" then return nil end
    return meshId .. "|" .. textureId
end

local function mapCache(cacheFolder, outMap, itemsList)
    if not cacheFolder then return end

    local assetLookup = {}
    local customLookup = {}
    local directLookup = {}
    for _, item in ipairs(itemsList) do
        if item.id ~= nil then assetLookup[tostring(item.id)] = item end
        if item.assetId ~= nil then assetLookup[tostring(item.assetId)] = item end
        if type(item.assetIds) == "table" then
            for _, assetId in ipairs(item.assetIds) do
                assetLookup[tostring(assetId)] = item
            end
        end
        if item.customModelName then customLookup[tostring(item.customModelName)] = item end
        if item.id ~= nil then directLookup[tostring(item.id)] = item end
        if item.phoneId ~= nil then directLookup[tostring(item.phoneId)] = item end
    end

    local function addMeshId(inst, targetItem, seenIds)
        local meshId = nil
        if inst:IsA("MeshPart") then
            meshId = inst.MeshId
        elseif inst:IsA("SpecialMesh") then
            meshId = inst.MeshId
        end
        if not meshId or meshId == "" then return end

        local normalized = normalizeContentId(meshId)
        local visualKey = getVisualKey(inst)
        if visualKey then
            if not VisualMap[visualKey] then VisualMap[visualKey] = {} end
            local exists = false
            for _, mapped in ipairs(VisualMap[visualKey]) do
                if mapped == targetItem or (mapped.name and targetItem.name and mapped.name:lower() == targetItem.name:lower()) then
                    exists = true
                    break
                end
            end
            if not exists then table.insert(VisualMap[visualKey], targetItem) end
        end

        if seenIds[normalized] then return end
        seenIds[normalized] = true

        if not outMap[normalized] then outMap[normalized] = {} end
        local exists = false
        for _, mapped in ipairs(outMap[normalized]) do
            if mapped == targetItem or (mapped.name and targetItem.name and mapped.name:lower() == targetItem.name:lower()) then
                exists = true
                break
            end
        end
        if not exists then table.insert(outMap[normalized], targetItem) end
    end

    for _, child in ipairs(cacheFolder:GetChildren()) do
        if child:IsA("Model") or child:IsA("Folder") or child:IsA("Accessory") then
            local assetIdStr = child.Name:match("ACC_(%d+)")
            local customName = child.Name:match("CUSTOM_(.+)")
            
            local targetItem = (assetIdStr and assetLookup[assetIdStr])
                or (customName and customLookup[customName])
                or directLookup[child.Name]
            
            if targetItem then
                local seenIds = {}
                addMeshId(child, targetItem, seenIds)
                for _, descendant in ipairs(child:GetDescendants()) do
                    if descendant:IsA("MeshPart") or descendant:IsA("SpecialMesh") then
                        addMeshId(descendant, targetItem, seenIds)
                    end
                end
            end
        end
    end
end

            local mapped = id ~= nil and MeshMap[tostring(id)] or nil
            if type(mapped) == "table" and mapped[1] then return mapped[1] end

        return nil

        if type(key) ~= "string" then return end
        local indexes = type(references) == "table" and references or {references}
        for _, index in ipairs(indexes) do
            local item = resolveExternalItem(index)
            if item then
                if not targetMap[key] then targetMap[key] = {} end
                local exists = false
                for _, mapped in ipairs(targetMap[key]) do
                    if mapped == item or (mapped.name and item.name and mapped.name:lower() == item.name:lower()) then
                        exists = true
                        break
                    end
                end
                if not exists then table.insert(targetMap[key], item) end
            end
        end

                    local name = type(candidate) == "table" and candidate.name or candidate

                    local item = normalizedName and NameMap[normalizedName] or nil
                    if item and not seenNames[normalizedName] then
                        seenNames[normalizedName] = true
                        table.insert(resolvedItems, item)
                    end
                end
            end
            if #resolvedItems == 1 then
                ClothingTemplateMap[templateType .. ":" .. normalizeContentId(templateId)] = resolvedItems[1]
            end
        end

        local profile = itemData.economyProfile or "normal"
        local rarity  = itemData.rarity or "Common"
        local price   = itemData.fairPrice or 0
        if price >= SOUND_ALERT_MIN_PRICE then
            if profile == "jackpot" then qualifies = true end
            if SOUND_ALERT_RARITIES[rarity] then qualifies = true end
            if SOUND_ALERT_JACKPOT_ONLY and profile ~= "jackpot" then qualifies = false end
        end
    end
    
    if not qualifies then return end
    AlertedObjects[obj] = true
    playAlertSound()

    local price   = itemData.fairPrice or 0
    notify("RARE ITEM!", string.format("%s — $%d [%s]", itemData.name or "?", math.floor(price), rarity), 6)

local function updateUserFly()
    local char = player.Character
    local root = char and (char:FindFirstChild("HumanoidRootPart") or char.PrimaryPart)
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not FLY_ENABLED or not root or not hum or isAutobuying then
        cleanupFlyMovers()
        if hum then hum.PlatformStand = false end
        return
    end

    hum.PlatformStand = true
    local camCF = camera.CFrame
    local velocity = Vector3.zero
    local rotation = camCF.Rotation

    if UserInputService:IsKeyDown(Enum.KeyCode.W) then
        velocity += camCF.LookVector
        rotation *= CFrame.Angles(math.rad(-40), 0, 0)
    end
    if UserInputService:IsKeyDown(Enum.KeyCode.S) then
        velocity -= camCF.LookVector
        rotation *= CFrame.Angles(math.rad(40), 0, 0)
    end
    if UserInputService:IsKeyDown(Enum.KeyCode.D) then
        velocity += camCF.RightVector
        rotation *= CFrame.Angles(0, 0, math.rad(-40))
    end
    if UserInputService:IsKeyDown(Enum.KeyCode.A) then
        velocity -= camCF.RightVector
        rotation *= CFrame.Angles(0, 0, math.rad(40))
    end
    if UserInputService:IsKeyDown(Enum.KeyCode.Space) then
        velocity += Vector3.yAxis
    end
    if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then
        velocity -= Vector3.yAxis
    end

    local bodyVelocity, bodyGyro = getFlyMovers(root)
    local tweenInfo = TweenInfo.new(0.5)
    TweenService:Create(bodyVelocity, tweenInfo, { Velocity = velocity * FLY_SPEED * 45 }):Play()
    TweenService:Create(bodyGyro, tweenInfo, { CFrame = rotation }):Play()
end

function floweringCanonicalItemRoot(obj)
    if not obj or obj == workspace then return nil end
    local droppedFolder = workspace:FindFirstChild("DroppedItems")
    local current = obj
    while current and current ~= workspace and current ~= game do
        local parent = current.Parent
        if current.Name:match("^M_%d+$") then return current end
        if current.Name:match("^W_%d+$") and parent == workspace then return current end
        if parent and (parent.Name == "ItemSlots" or parent.Name == "AccessorySlots") then return current end
        if droppedFolder and parent == droppedFolder then return current end
        current = parent
    end
    return nil
end

function floweringNormalizeItemText(value)
    return tostring(value or "")
        :gsub("<.->", " ")
        :gsub("[%c]", " ")
        :gsub("%s+", " ")
        :match("^%s*(.-)%s*$")
        :lower()
end

function floweringLinkedVisualRoot(root)
    if not root then return nil end
    local current = root
    while current and current ~= workspace do
        for _, child in ipairs(current:GetChildren()) do
            if child:IsA("ObjectValue") and child.Name:lower():find("shopvisualref", 1, true) and child.Value then
                return child.Value
            end
        end
        if current.Name:find("Shop_ShopZone_", 1, true) == 1 then break end
        current = current.Parent
    end
    for _, descendant in ipairs(root:GetDescendants()) do
        if descendant:IsA("ObjectValue") and descendant.Name:lower():find("shopvisualref", 1, true) and descendant.Value then
            return descendant.Value
        end
    end
    return nil
end

function floweringResolveItemFromText(root)
    if not root then return nil end
    local texts = {}
    local function addText(value)
        local normalized = floweringNormalizeItemText(value)
        if normalized ~= "" then table.insert(texts, normalized) end
    end
    local function matchTexts()
        for _, text in ipairs(texts) do
            local exact = NameMap[text]
            if exact then return exact end
            for _, entry in ipairs(NormalizedNameEntries) do
                if #entry.name >= 4 and text:find(entry.name, 1, true) then
                    return entry.item
                end
            end
        end
        return nil
    end

    for attributeName, value in pairs(root:GetAttributes()) do
        if type(value) == "string" then
            local key = tostring(attributeName):lower()
            if key:find("name", 1, true) or key:find("item", 1, true) then addText(value) end
        end
    end
    for _, descendant in ipairs(root:GetDescendants()) do
        if descendant:IsA("TextLabel") or descendant:IsA("TextButton") or descendant:IsA("TextBox") then
            addText(descendant.Text)
        end
    end
    local internalMatch = matchTexts()
    if internalMatch then return internalMatch end

    return nil
end

local function buildItemDataFromSlotInfo(slotInfo)
    if type(slotInfo) ~= "table" then return nil end
    local name = tostring(slotInfo.name or "")
    if name == "" then return nil end

    local configured = NameMap[name:lower()]
    local itemData = {}
    if type(configured) == "table" then
        for key, value in pairs(configured) do
            itemData[key] = value
        end
    end

    itemData.name = name
    itemData.rarity = slotInfo.rarity or itemData.rarity or "Unknown"
    itemData.fairPrice = itemData.fairPrice or itemData.price or tonumber(slotInfo.price)
    itemData.shopPrice = tonumber(slotInfo.price)
    itemData.economyProfile = itemData.economyProfile or "normal"
    itemData.slotId = slotInfo.slotId
    itemData.fromSlotInfo = true
    return itemData
end

local function AddItemToCache(obj)
    pcall(function()
        if not obj then return end

        local sourceObj = obj
        local droppedFolder = workspace:FindFirstChild("DroppedItems")
        obj = floweringCanonicalItemRoot(obj)
        if not obj then return end
        local slotInfo = ShopSlotInfoByRoot[obj]
        local slotVisual = slotInfo and slotInfo.visualRef
        local visualRoot = (typeof(slotVisual) == "Instance" and slotVisual.Parent and slotVisual)
            or floweringLinkedVisualRoot(obj)
            or obj
        if CachedItems[obj] then
            if slotInfo then return end
            if not (sourceObj:IsA("TextLabel") or sourceObj:IsA("TextButton") or sourceObj:IsA("TextBox") or sourceObj:IsA("BillboardGui") or sourceObj:IsA("SurfaceGui")) then
                return
            end
            local correctedItem = floweringResolveItemFromText(obj)
            if correctedItem and CachedItems[obj].Data ~= correctedItem then
                local cached = CachedItems[obj]
                cached.Data = correctedItem
                local correctedIsRainbow = correctedItem.economyProfile == "jackpot"
                local correctedColor = correctedIsRainbow and Color3.fromHSV((tick() * 0.18) % 1, 1, 1) or pColor(correctedItem.rarity)
                cached.Color = correctedColor
                cached.IsRainbow = correctedIsRainbow
                if cached.Gradient then
                    cached.Gradient.Color = correctedIsRainbow and RAINBOW_COLOR_SEQUENCE or ColorSequence.new({
                            ColorSequenceKeypoint.new(0, correctedColor),
                            ColorSequenceKeypoint.new(0.5, Color3.new(1, 1, 1)),
                            ColorSequenceKeypoint.new(1, correctedColor),
                        })
                end
            end
            return
        end
        local isInDropped = false
        local inShopZone = false

        local ancestor = obj
        while ancestor and ancestor ~= workspace and ancestor ~= game do
            if ancestor:IsA("Model") and Players:GetPlayerFromCharacter(ancestor) then return end
            if droppedFolder and ancestor == droppedFolder then isInDropped = true end
            if ancestor.Parent == workspace and ancestor.Name:match("^W_%d+$") then isInDropped = true end
            if string.find(ancestor.Name, "Shop_ShopZone_") == 1 then inShopZone = true end
            ancestor = ancestor.Parent
        end

        if not isInDropped and not inShopZone then return end

        local possibleIds = {}
        local possibleVisualKeys = {}
        local possibleClothingTemplates = {}
        local hasWearableAsset = false
        local function addId(inst)
            if inst:IsA("MeshPart") or inst:IsA("SpecialMesh") then
                if inst.MeshId ~= "" then table.insert(possibleIds, normalizeContentId(inst.MeshId)) end
                local visualKey = getVisualKey(inst)
                if visualKey then table.insert(possibleVisualKeys, visualKey) end
            elseif inst:IsA("Shirt") then
                hasWearableAsset = true
                if inst.ShirtTemplate ~= "" then
                    local templateId = normalizeContentId(inst.ShirtTemplate)
                    table.insert(possibleIds, templateId)
                    table.insert(possibleClothingTemplates, {type = "Shirt", id = templateId})
                end
            elseif inst:IsA("Pants") then
                hasWearableAsset = true
                if inst.PantsTemplate ~= "" then
                    local templateId = normalizeContentId(inst.PantsTemplate)
                    table.insert(possibleIds, templateId)
                    table.insert(possibleClothingTemplates, {type = "Pants", id = templateId})
                end
            elseif inst:IsA("Decal") then
                hasWearableAsset = true
                if inst.Texture ~= "" then table.insert(possibleIds, normalizeContentId(inst.Texture)) end
            end
        end

        if visualRoot:IsA("Model") or visualRoot:IsA("BasePart") then
            for _, child in ipairs(visualRoot:GetDescendants()) do addId(child) end
        end
        addId(visualRoot)

        local detectedItem = buildItemDataFromSlotInfo(slotInfo) or floweringResolveItemFromText(obj)
        if not detectedItem then
            for _, clothingTemplate in ipairs(possibleClothingTemplates) do
                detectedItem = ClothingTemplateMap[clothingTemplate.type .. ":" .. clothingTemplate.id]
                if detectedItem then break end
            end
        end
        if not detectedItem then
        for _, visualKey in ipairs(possibleVisualKeys) do
            local visualItems = VisualMap[visualKey]
            if visualItems and #visualItems == 1 then
                detectedItem = visualItems[1]
                break
            end
        end
        end

        if not detectedItem then
        for _, rawId in ipairs(possibleIds) do
            local mId = normalizeContentId(rawId)
            local numberMatch = string.match(mId, "%d+")
            local items = MeshMap[mId] or (numberMatch and NumericMeshMap[numberMatch])
            if items then
                    if #items == 1 then
                        detectedItem = items[1]
                    else
                        local objName = obj.Name:lower()
                        local parentName = obj.Parent and obj.Parent.Name:lower() or ""
                        for _, item in ipairs(items) do
                            local iName = item.name:lower()
                            if objName == iName or parentName == iName or string.find(parentName, iName, 1, true) then
                                detectedItem = item
                                break
                            end
                        end
                    end
            end
            if detectedItem then break end
        end
        end

        if not detectedItem then
            local n = obj.Name:lower()
            if NameMap[n] then detectedItem = NameMap[n] end
        end

        if not detectedItem then
            local isUnknownAccessory = visualRoot:GetAttribute("__ShopVisualRoot") == true and #possibleIds > 0
            if not isUnknownAccessory and not hasWearableAsset then return end
            detectedItem = {
                name = "Unknown " .. tostring(visualRoot.Name),
                rarity = "Unknown",
                spawnChance = 0,
                economyProfile = "normal",
                isUnknown = true,
            }
        end

        local posType = 0
        local position = nil
        local posObj = visualRoot
        if visualRoot:IsA("Shirt") or visualRoot:IsA("Pants") or visualRoot:IsA("Decal") then
            posObj = visualRoot.Parent
        end

        if posObj and posObj:IsA("BasePart") then
            posType = 1
            position = posObj.Position
        elseif posObj and posObj:IsA("Model") then
            if posObj.PrimaryPart then
                posType = 2
                position = posObj.PrimaryPart.Position
            else
                posType = 3
                local success, result = pcall(function() return posObj:GetPivot().Position end)
                if success then position = result end
            end
        end
        if not position then return end

        if not slotInfo then
            for _, cData in pairs(CachedItems) do
                if cData.Data and cData.Data.name == detectedItem.name and cData.Position then
                    if (cData.Position - position).Magnitude < 5 then return end
                end
            end
        end

        local label = Instance.new("TextLabel")
        label.BackgroundTransparency = 1
        label.Font = Enum.Font.GothamBold
        label.TextColor3 = Color3.new(1, 1, 1)
        label.RichText = true
        label.AnchorPoint = Vector2.new(0.5, 1)
        label.AutomaticSize = Enum.AutomaticSize.XY
        label.Visible = false
        label.Parent = espScreenGui

        local labelStroke = Instance.new("UIStroke")
        labelStroke.Thickness = 1
        labelStroke.Parent = label

        local gradient = Instance.new("UIGradient")
        gradient.Rotation = 0
        gradient.Parent = label

        local targetMesh = posObj or obj
        local mannequin = nil
        if targetMesh.Name == "Mannequin" then mannequin = targetMesh
        elseif targetMesh:FindFirstChild("Mannequin") then mannequin = targetMesh:FindFirstChild("Mannequin")
        elseif targetMesh.Parent and targetMesh.Parent:FindFirstChild("Mannequin") then mannequin = targetMesh.Parent:FindFirstChild("Mannequin")
        end
        if mannequin then
            targetMesh = mannequin
        elseif targetMesh:IsA("Model") then
            for _, child in ipairs(targetMesh:GetDescendants()) do
                if child:IsA("MeshPart") or (child:IsA("Part") and child:FindFirstChildOfClass("SpecialMesh")) then
                    targetMesh = child
                    break
                end
            end
            if targetMesh == (posObj or obj) and targetMesh.PrimaryPart then targetMesh = targetMesh.PrimaryPart end
        end

        local yOffset = 0
        if targetMesh:IsA("Model") then
            local success, size = pcall(function() return targetMesh:GetExtentsSize() end)
            if success then yOffset = size.Y / 2 end
        elseif targetMesh:IsA("BasePart") then
            yOffset = targetMesh.Size.Y / 2
        end

        local economyProfile = detectedItem.economyProfile or "normal"
        local isRainbow = economyProfile == "jackpot"
        local col
        if isRainbow then
            col = Color3.fromHSV((tick() * 0.18) % 1, 1, 1)
        else
            col = pColor(detectedItem.rarity)
        end
        
        local colorSeq = isRainbow and RAINBOW_COLOR_SEQUENCE or ColorSequence.new({
                ColorSequenceKeypoint.new(0, col),
                ColorSequenceKeypoint.new(0.5, Color3.new(1, 1, 1)),
                ColorSequenceKeypoint.new(1, col),
            })
        gradient.Color = colorSeq

        CachedItems[obj] = {
            Data = detectedItem,
            Label = label,
            Gradient = gradient,
            TargetMesh = targetMesh,
            VisualRoot = visualRoot,
            PosType = posType,
            Position = position,
            YOffset = yOffset,
            Color = col,
            IsRainbow = isRainbow,
            IsDropped = isInDropped,
            ShouldShow = false,
            Text = "",
        }

        if not isInDropped then
            checkAlert(obj, detectedItem)
        end
    end)
end

local function refreshShopSlot(slotInfo)
    if type(slotInfo) ~= "table" or typeof(slotInfo.slotRef) ~= "Instance" then return end
    local root = floweringCanonicalItemRoot(slotInfo.slotRef)
    if not root then return end

    local previous = ShopSlotInfoByRoot[root]
    local unchanged = previous
        and previous.slotId == slotInfo.slotId
        and previous.name == slotInfo.name
        and previous.rarity == slotInfo.rarity
        and previous.price == slotInfo.price
        and previous.visualRef == slotInfo.visualRef

    ShopSlotInfoByRoot[root] = slotInfo
    if not ShopSlotVisualConnections[root] then
        local visualReference = slotInfo.slotRef:FindFirstChild("__ShopVisualRef")
        if visualReference and visualReference:IsA("ObjectValue") then
            local connection = visualReference:GetPropertyChangedSignal("Value"):Connect(function()
                if SCRIPT_UNLOADED then return end
                local currentInfo = ShopSlotInfoByRoot[root]
                if currentInfo and currentInfo.visualRef ~= visualReference.Value then
                    ShopSlotInfoByRoot[root] = nil
                    if CachedItems[root] then RemoveItemFromCache(root) end
                    if root.Parent then task.defer(AddItemToCache, root) end
                end
            end)
            ShopSlotVisualConnections[root] = connection
            trackConnection(connection)
        end
    end

    if unchanged and CachedItems[root] then return end
    if CachedItems[root] then RemoveItemFromCache(root) end
    task.defer(AddItemToCache, root)
end

local function attachShopSlotFeed()
    local shopRemotes = RS:FindFirstChild("ShopRemotes")
    if not shopRemotes then return end

    local slotInfoUpdate = shopRemotes:FindFirstChild("SlotInfoUpdate")
    if slotInfoUpdate and slotInfoUpdate:IsA("RemoteEvent") then
        trackConnection(slotInfoUpdate.OnClientEvent:Connect(function(payload)
            if SCRIPT_UNLOADED or type(payload) ~= "table" then return end
            for _, slotInfo in ipairs(payload.slots or {}) do
                refreshShopSlot(slotInfo)
            end
        end))
    end
end

local UI_CTRLS = {}

local function safeSet(element, value)
    if not element then return end
    pcall(function()
        if element.Class == "Toggle" then
            if element.Set then element:Set({ CurrentValue = value == true }) end
        elseif element.Class == "Slider" then
            if element.Set then element:Set({ CurrentValue = tonumber(value) or value }) end
        elseif element.Class == "Input" then
            if element.Set then element:Set({ CurrentValue = tostring(value or "") }) end
        elseif element.Class == "Dropdown" then
            if element.Set then element:Set({ CurrentOption = value }) end
        elseif element.Class == "Colorpicker" then
            if element.Set then element:Set({ Color = value }) end
        elseif element.Select and type(element.Select) == "function" then
            element:Select(value)
        elseif element.Set then
            element:Set({ CurrentValue = value, CurrentOption = value, Color = value })
        elseif element.SetValue then
            element:SetValue(value)
        end
    end)
end

local function refreshDropdownOptions(element, options, currentOption)
    if not element or not element.Set then return end
    pcall(function()
        element:Set({
            Options = options or {},
            CurrentOption = currentOption or {},
        })
    end)
end

local function normalizeConfigOptions(files)
    if type(files) ~= "table" or #files == 0 then
        return {"No configs"}
    end
    return files
end

local function refreshConfigDropdown(files, selectedName)
    local options = normalizeConfigOptions(files)
    local current = "No configs"
    if selectedName and selectedName ~= "" then
        local wanted = selectedName:match("%.json$") and selectedName or (selectedName .. ".json")
        for _, file in ipairs(options) do
            if file == wanted then
                current = wanted
                break
            end
        end
    elseif options[1] ~= "No configs" then
        current = options[1]
    end
    refreshDropdownOptions(UI_CTRLS.ConfigDropdown, options, current)
end

local function colorToHex(color)
    return string.format("#%02X%02X%02X", math.floor(color.R * 255), math.floor(color.G * 255), math.floor(color.B * 255))
end

local function hexToColor3(hex)
    if type(hex) ~= "string" then return nil end
    local r, g, b = hex:match("^#?(%x%x)(%x%x)(%x%x)$")
    if not r then return nil end
    return Color3.fromRGB(tonumber(r, 16), tonumber(g, 16), tonumber(b, 16))
end

local function loadConfig(name)

    local folderName = "flowering"

    local data = nil
    pcall(function()
        if isfile and isfile(filename) then
            data = readfile(filename)
        end
    end)
    
    if not data then
        notify("Config System", "Конфиг не найден или не прочитан!", 3)
        return
    end
    
    local success, config = pcall(function()
        return HttpService:JSONDecode(data)
    end)
    
    if not success or type(config) ~= "table" then
        notify("Config System", "Ошибка разбора JSON!", 3)

local function saveAutoload(name)
    local folderName = "flowering"
    pcall(function()
        if writefile and isfolder then
            if not isfolder(folderName) then
                makefolder(folderName)
            end
            writefile(folderName .. "/autoload.txt", name or "")
        end
    end)
end

local function getAutoload()
    local folderName = "flowering"
    local autoloadFile = folderName .. "/autoload.txt"
    local name = nil
    pcall(function()
        if isfile and isfile(autoloadFile) then
            name = readfile(autoloadFile)
        end
    end)
    return name
end

local function multiToDict(values)
    local out = {}
    if type(values) ~= "table" then return out end
    for k, v in pairs(values) do
        if type(k) == "string" and v == true then
            out[k] = true
        elseif type(k) == "string" and type(v) == "string" then
            out[k] = true
        elseif type(k) == "number" and type(v) == "string" then
            out[v] = true
        elseif type(v) == "table" then
            if v.Name and v.Value ~= false then out[tostring(v.Name)] = true end
            if v.Title and v.Value ~= false then out[tostring(v.Title)] = true end
        end
    end
    return out
end

local function arrCopy(t)
    local out = {}
    for i, v in ipairs(t) do out[i] = v end
    return out
end

local function getPositionOfInstance(inst)
    if not inst then return Vector3.new(0,0,0) end
    if inst:IsA("BasePart") then
        return inst.Position
    elseif inst:IsA("Model") then
        if inst.PrimaryPart then
            return inst.PrimaryPart.Position
        end
        local ok, pos = pcall(function() return inst:GetPivot().Position end)
        if ok and pos then return pos end
    end
    local parent = inst.Parent
    if parent and parent:IsA("BasePart") then
        return parent.Position
    end
    return Vector3.new(0,0,0)
end

local function firePrompt(p, lookTarget)
    if not p then return end
    local char = player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    
    local targetPart = p.Parent
    if targetPart and targetPart:IsA("Attachment") then
        targetPart = targetPart.Parent
    end
    if not targetPart then return end
    
    local targetPos = getPositionOfInstance(targetPart)
    if not targetPos or targetPos == Vector3.new(0,0,0) then return end
    
    pcall(function()
        local direction = (hrp.Position - targetPos).Unit
        if direction.Magnitude < 0.1 or tostring(direction.X) == "nan" then
            direction = hrp.CFrame.LookVector
        end
        local teleportPos = targetPos + direction * 1.8
        hrp.CFrame = CFrame.lookAt(teleportPos, targetPos)
    end)
    
    local lookChanged = false
    if AUTO_BUY_CAMERA_LOOK then
        pcall(function()
            local lookAtPos = getPositionOfInstance(lookTarget or targetPart)
            camera.CameraType = Enum.CameraType.Scriptable
            camera.CFrame = CFrame.lookAt(hrp.Position + Vector3.new(0, 2.0, 0), lookAtPos)
            lookChanged = true
        end)
        task.wait(0.08)
    end
    
    pcall(function()
        if p.Enabled then
            if fireproximityprompt then
                fireproximityprompt(p)
            else
                p:InputHoldBegin()
                task.wait(p.HoldDuration + 0.02)
                p:InputHoldEnd()
            end
        end
    end)
    
    if lookChanged then
        pcall(function()
            camera.CameraType = Enum.CameraType.Custom
        end)
    end
end

local function findClickDetector(inst, depth)
    if not inst then return nil end
    if inst:IsA("ClickDetector") then return inst end
    local cd = inst:FindFirstChildWhichIsA("ClickDetector", true)
    if cd then return cd end
    local anc = inst.Parent
    for _ = 1, (depth or 3) do
        if not anc or anc == workspace or anc == game then break end
        cd = anc:FindFirstChildWhichIsA("ClickDetector", true)
        if cd then return cd end
        anc = anc.Parent
    end
    return nil
end

local function fireClick(cd)
    if not cd or not cd.Parent then return false end
    local char = player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return false end

    local targetPart = cd.Parent
    if not targetPart:IsA("BasePart") then
        targetPart = targetPart:FindFirstChildWhichIsA("BasePart") or targetPart:FindFirstChildWhichIsA("BasePart", true)
    end
    local targetPos = targetPart and targetPart.Position or getPositionOfInstance(cd.Parent)
    if not targetPos then return false end

    local maxDist = cd.MaxActivationDistance or 12
    if maxDist <= 0 then maxDist = 12 end
    if (hrp.Position - targetPos).Magnitude > maxDist - 2 then
        pcall(function()
            local dir = (hrp.Position - targetPos).Unit
            if dir.X ~= dir.X then dir = hrp.CFrame.LookVector end
            local dist = math.max(math.min(maxDist - 4, 6), 2)
            hrp.CFrame = CFrame.lookAt(targetPos + dir * dist, targetPos)
        end)
        task.wait(0.1)
    end

    local ok = false
    if fireclickdetector then
        ok = pcall(fireclickdetector, cd)
    end
    if not ok then
        pcall(function()
            camera.CameraType = Enum.CameraType.Scriptable
            camera.CFrame = CFrame.lookAt(hrp.Position + Vector3.new(0, 2, 0), targetPos)
        end)
        task.wait(0.08)
        pcall(function()
            local sp, onScreen = camera:WorldToViewportPoint(targetPos)
            if onScreen then
                VIM:SendMouseMoveEvent(sp.X, sp.Y, game)
                task.wait(0.04)
                VIM:SendMouseButtonEvent(sp.X, sp.Y, 0, true, game, 1)
                task.wait(0.04)
                VIM:SendMouseButtonEvent(sp.X, sp.Y, 0, false, game, 1)
                ok = true
            end
        end)
        pcall(function()
            camera.CameraType = Enum.CameraType.Custom
        end)
    end
    return ok
end

local function interactWith(obj, lookTarget)
    if not obj then return false end
    local cd = findClickDetector(obj)
    if cd then
        local ok = fireClick(cd)
        if ok then return true end
    end
    local p = obj:IsA("ProximityPrompt") and obj or obj:FindFirstChildWhichIsA("ProximityPrompt", true)
    if not p and obj.Parent then
        p = obj.Parent:FindFirstChildWhichIsA("ProximityPrompt", true)
    end
    if p and p.Parent then
        firePrompt(p, lookTarget)
        return true
    end
    return false
end

local function findCashierPrompt(shopZone, mannequinPos)
    if shopZone then
        for _, desc in ipairs(shopZone:GetDescendants()) do
            if desc:IsA("ProximityPrompt") then
                local actionText = desc.ActionText:lower()
                local objectText = desc.ObjectText:lower()
                
                local isCashier = false
                if string.find(actionText, "pay") or string.find(actionText, "buy") or string.find(actionText, "purchase") or
                   string.find(actionText, "checkout") or string.find(actionText, "оплат") or string.find(actionText, "куп") or
                   string.find(actionText, "касс") or string.find(actionText, "подтверд") or
                   string.find(objectText, "seller") or string.find(objectText, "cashier") or string.find(objectText, "vendor") or
                   string.find(objectText, "npc") or string.find(objectText, "checkout") or string.find(objectText, "register") or
                   string.find(objectText, "продав") or string.find(objectText, "кассир") then
                    isCashier = true
                end
                
                if not isCashier then
                    local anc = desc.Parent
                    for depth = 1, 5 do
                        if not anc or anc == shopZone or anc == workspace then break end
                        local ancName = anc.Name:lower()
                        if string.find(ancName, "seller") or string.find(ancName, "cashier") or string.find(ancName, "vendor") or
                           string.find(ancName, "npc") or string.find(ancName, "checkout") or string.find(ancName, "register") or
                           string.find(ancName, "продав") or string.find(ancName, "кассир") or string.find(ancName, "cash") or
                           string.find(ancName, "оплат") or string.find(ancName, "касс") then
                            isCashier = true
                            break
                        end
                        anc = anc.Parent
                    end
                end
                
                if isCashier then
                    return desc
                end
            end
        end
    end
    
    if mannequinPos then
        local closestPrompt = nil
        local minDist = 80
        for _, desc in ipairs(workspace:GetDescendants()) do
            if desc:IsA("ProximityPrompt") and desc.Parent then
                local actionText = desc.ActionText:lower()
                local objectText = desc.ObjectText:lower()
                
                local isCashier = false
                if string.find(actionText, "pay") or string.find(actionText, "buy") or string.find(actionText, "checkout") or
                   string.find(actionText, "оплат") or string.find(actionText, "куп") or string.find(actionText, "касс") or
                   string.find(objectText, "seller") or string.find(objectText, "cashier") or string.find(objectText, "register") or
                   string.find(objectText, "продав") or string.find(objectText, "кассир") then
                    isCashier = true
                end
                
                if not isCashier then
                    local anc = desc.Parent
                    for depth = 1, 4 do
                        if not anc or anc == workspace then break end
                        local ancName = anc.Name:lower()
                        if string.find(ancName, "seller") or string.find(ancName, "cashier") or string.find(ancName, "vendor") or
                           string.find(ancName, "npc") or string.find(ancName, "checkout") or string.find(ancName, "register") or
                           string.find(ancName, "продав") or string.find(ancName, "кассир") or string.find(ancName, "cash") or
                           string.find(ancName, "оплат") or string.find(ancName, "касс") then
                            isCashier = true
                            break
                        end
                        anc = anc.Parent
                    end
                end
                
                if isCashier then
                    local pPart = desc.Parent
                    if pPart:IsA("Attachment") then pPart = pPart.Parent end
                    if pPart and pPart:IsA("BasePart") then
                        local dist = (pPart.Position - mannequinPos).Magnitude
                        if dist < minDist then
                            minDist = dist
                            closestPrompt = desc
                        end
                    end
                end
            end
        end
        if closestPrompt then return closestPrompt end
    end
    
    if shopZone then
        for _, desc in ipairs(shopZone:GetDescendants()) do
            if desc:IsA("ProximityPrompt") then
                local parentName = desc.Parent and desc.Parent.Name:lower() or ""
                if not string.find(parentName, "mannequin") and not string.find(parentName, "stand") and 
                   not string.find(parentName, "вещ") and not string.find(parentName, "cloth") and 
                   not string.find(parentName, "shirt") and not string.find(parentName, "pants") and
                   not string.find(parentName, "hanger") then
                    return desc
                end
            end
        end
    end
    return nil
end

        return pPart
    end
    return nil
end

        tostring(nextState),

local function autoBuyFail(context, message)
    message = tostring(message or "Неизвестная ошибка")
    local failedStage = AUTO_BUY_STATE_LABELS[context.state] or tostring(context.state)
    local detailedMessage = failedStage .. ": " .. message
    context.error = detailedMessage
    AutoBuyMachine.LastError = detailedMessage

    local target = context.target
    if target and target.obj and target.obj.Parent and not context.itemInteracted then
        AutoBuyRetryAfter[target.obj] = tick() + AUTO_BUY_RETRY_DELAY
    end

    warn(string.format("[Flowering Auto Buy] %s", detailedMessage))
    log_write(string.format("Auto Buy error: %s", detailedMessage))
    notify("Auto Buy · " .. failedStage, message, 3)
    return AUTO_BUY_STATE.FAILED, detailedMessage
end

local function findNearestAutoBuyTarget(hrp)
    local nearestTarget = nil
    local minDist = math.huge
    local now = tick()

    for obj, cData in pairs(CachedItems) do
        if not obj.Parent or cData.IsDropped or BoughtItems[obj] then continue end
        if (AutoBuyRetryAfter[obj] or 0) > now then continue end

        local itemData = cData.Data
        if not itemData then continue end

        local rarity = itemData.rarity or "Common"
        local economy = itemData.economyProfile or "normal"
        if AUTO_BUY_RARITIES[rarity] == true and AUTO_BUY_ECONOMY[economy] == true then
            local position = cData.Position
            if position then
                local distance = (position - hrp.Position).Magnitude
                if distance < minDist then
                    minDist = distance
                    nearestTarget = {obj = obj, pos = position, data = itemData}
                end
            end
        end
    end

    return nearestTarget
end

AutoBuyStateHandlers[AUTO_BUY_STATE.SEARCH] = function(context)
    local hrp = getHRP()
    if not hrp then return autoBuyFail(context, "Персонаж ещё не готов") end

    local target = findNearestAutoBuyTarget(hrp)
    if not target then return AUTO_BUY_STATE.COMPLETE, "Подходящих предметов нет" end

    context.target = target
    context.shopZone = findTargetShopZone(target)
    context.travelMethodName = AUTO_BUY_METHOD == "Walk" and "Идем" or "Летим"
    notify(
        "Auto Buy",
        context.travelMethodName .. " к: " .. (target.data.name or "шмотке") .. " [" .. (target.data.rarity or "Common") .. "]",
        2
    )
    return AUTO_BUY_STATE.MOVE_TO_ITEM,
        "Цель: " .. (target.data.name or "неизвестный предмет") .. " [" .. (target.data.rarity or "Common") .. "]"
end

AutoBuyStateHandlers[AUTO_BUY_STATE.MOVE_TO_ITEM] = function(context)
    local target = context.target
    if not target or not target.obj.Parent then
        return autoBuyFail(context, "Предмет исчез до начала движения")
    end
    if not travelTo(target.pos, AUTO_BUY_SPEED, 5.5) then
        if not autoBuyCanContinue() then
            return AUTO_BUY_STATE.STOPPED, "Автовыкуп остановлен во время движения к предмету."
        end
        return autoBuyFail(context, "Не удалось добраться до предмета")
    end
    if not target.obj.Parent then
        return autoBuyFail(context, "Предмет исчез во время движения")
    end
    return AUTO_BUY_STATE.INTERACT_ITEM, "Предмет достигнут, выполняю взаимодействие."
end

AutoBuyStateHandlers[AUTO_BUY_STATE.INTERACT_ITEM] = function(context)
    local target = context.target
    task.wait(0.15)

    local cached = CachedItems[target.obj]
    local targetMesh = cached and cached.TargetMesh or target.obj
    local interacted = interactWith(target.obj, targetMesh)
    if not interacted and target.obj.Parent then
        interacted = interactWith(target.obj.Parent, targetMesh)
    end
    if not interacted then
        return autoBuyFail(context, "Не удалось взаимодействовать с предметом")
    end

    context.itemInteracted = true
    BoughtItems[target.obj] = true
    task.wait(0.25)

    context.seller = findSeller(context.shopZone, target.pos)
    if not context.seller then
        return autoBuyFail(context, "Продавец в этой зоне не найден")
    end
    context.sellerPos = getSellerPosition(context.seller)
    if not context.sellerPos then
        return autoBuyFail(context, "Не удалось определить позицию продавца")
    end
    return AUTO_BUY_STATE.MOVE_TO_SELLER, "Предмет добавлен, направляюсь к продавцу."
end

AutoBuyStateHandlers[AUTO_BUY_STATE.MOVE_TO_SELLER] = function(context)
    notify("Auto Buy", context.travelMethodName .. " к продавцу для оплаты...", 2)
    if not travelTo(context.sellerPos, AUTO_BUY_SPEED, 5.5) then
        if not autoBuyCanContinue() then
            return AUTO_BUY_STATE.STOPPED, "Автовыкуп остановлен во время движения к продавцу."
        end
        return autoBuyFail(context, "Не удалось добраться до продавца")
    end
    return AUTO_BUY_STATE.CLICK_SELLER, "Продавец достигнут, выполняю клик."
end

AutoBuyStateHandlers[AUTO_BUY_STATE.CLICK_SELLER] = function(context)
    task.wait(0.15)
    if not interactWith(context.seller) then
        return autoBuyFail(context, "Не удалось нажать на продавца")
    end
    task.wait(0.35)
    return AUTO_BUY_STATE.COMPLETE, "Продавец нажат, покупка отправлена."
end

local function runAutoBuyMachine()
    local context = {state = AUTO_BUY_STATE.SEARCH}
    AutoBuyMachine.Context = context
    AutoBuyMachine.State = context.state
    AutoBuyMachine.LastError = nil
    autoBuySetStatus(context.state)

    local steps = 0
    while context.state ~= AUTO_BUY_STATE.COMPLETE
        and context.state ~= AUTO_BUY_STATE.FAILED
        and context.state ~= AUTO_BUY_STATE.STOPPED do
        if not autoBuyCanContinue() then
            autoBuyTransition(context, AUTO_BUY_STATE.STOPPED, "Автовыкуп остановлен пользователем.")
            break
        end

        steps = steps + 1
        if steps > 12 then
            local failedState, reason = autoBuyFail(context, "Превышен лимит переходов автомата")
            autoBuyTransition(context, failedState, reason)
            break
        end

        local handler = AutoBuyStateHandlers[context.state]
        if not handler then
            local failedState, reason = autoBuyFail(context, "Нет обработчика состояния " .. tostring(context.state))
            autoBuyTransition(context, failedState, reason)
            break
        end

        local ok, nextState, reason = xpcall(function()
            return handler(context)
        end, autoBuyTraceback)
        if not ok then
            nextState, reason = autoBuyFail(context, nextState)
        end
        autoBuyTransition(context, nextState, reason)
    end

    local completed = context.state == AUTO_BUY_STATE.COMPLETE
    AutoBuyMachine.Context = nil
    AutoBuyMachine.State = AUTO_BUY_STATE.IDLE
    return completed, context.error
end

task.spawn(function()
    while not SCRIPT_UNLOADED do
        task.wait(0.5)
        if SCRIPT_UNLOADED then break end
        if AUTO_BUY_ENABLED and not isAutobuying and AUTH_PASSED then
            isAutobuying = true
            local ok, err = xpcall(runAutoBuyMachine, autoBuyTraceback)
            isAutobuying = false
            if not ok then
                AutoBuyMachine.Context = nil
                AutoBuyMachine.State = AUTO_BUY_STATE.IDLE
                AutoBuyMachine.LastError = tostring(err)
                autoBuySetStatus(AUTO_BUY_STATE.FAILED, "Критическая ошибка: " .. tostring(err))
                warn("[Flowering Auto Buy] Критическая ошибка автомата: " .. tostring(err))
                log_write("Auto Buy critical error: " .. tostring(err))
                notify("Auto Buy", "Критическая ошибка, подробности в консоли", 3)
            end
        end
    end
end)

local function destroyInterface()
    pcall(function()
        if Luna and Luna.Destroy then
            Luna:Destroy()
        end
    end)

    pcall(function()
        if Window then
            if Window.Destroy then
                Window:Destroy()
            elseif Window.Unload then
                Window:Unload()
            elseif Window.Close then
                Window:Close()
            end
        end
    end)

    local roots = {}
    local function addRoot(root)
        if root then table.insert(roots, root) end
    end

    pcall(function() addRoot(game:GetService("CoreGui")) end)
    pcall(function() addRoot(player and player:FindFirstChildOfClass("PlayerGui")) end)
    pcall(function()
        if gethui then addRoot(gethui()) end
    end)

    local markers = {
        "tsum flowering",
        "inputfuck",
        "interface hidden",
        "luna",
        "millenium",
        "windui"
    }

    local function textMatches(value)
        value = tostring(value or ""):lower()
        for _, marker in ipairs(markers) do
            if value:find(marker, 1, true) then
                return true
            end
        end
        return false
    end

    local function destroyGuiObject(inst, root)
        local target = inst
        while target and target.Parent and target.Parent ~= root do
            if target:IsA("ScreenGui") then
                break
            end
            target = target.Parent
        end
        if target then
            pcall(function() target:Destroy() end)
        end
    end

    for _, root in ipairs(roots) do
        pcall(function()
            for _, inst in ipairs(root:GetDescendants()) do
                if textMatches(inst.Name) then
                    destroyGuiObject(inst, root)
                elseif (inst:IsA("TextLabel") or inst:IsA("TextButton") or inst:IsA("TextBox")) and textMatches(inst.Text) then
                    destroyGuiObject(inst, root)
                end
            end

            for _, inst in ipairs(root:GetChildren()) do
                if textMatches(inst.Name) then
                    pcall(function() inst:Destroy() end)
                end
            end
        end)
    end
end
